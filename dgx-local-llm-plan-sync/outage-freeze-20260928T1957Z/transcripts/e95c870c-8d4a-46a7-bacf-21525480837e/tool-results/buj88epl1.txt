     1	/**
     2	 * Ploinky agent OpenAI backend module (`adapter_key = ploinky-agent-openai`).
     3	 *
     4	 * Calls a discovered Ploinky agent's `/v1/chat/completions` THROUGH the Ploinky
     5	 * router, authenticating with an HTTP Agent Assertion that Soul Gateway signs
     6	 * with its own injected `PLOINKY_AGENT_SECRET`. Providers for these agents are
     7	 * written by the agent-discovery reconciler with `auth_strategy = 'none'`: no
     8	 * upstream credential is leased; this backend self-signs every call.
     9	 *
    10	 * ── Why this backend owns its HTTP transport (documented decision) ──────────
    11	 *
    12	 * Soul Gateway's convention is that request-time LLM inference goes through
    13	 * `achillesAgentLib` and LLM `execute()` paths do not add ad-hoc upstream HTTP.
    14	 * That rule targets VENDOR LLM inference (OpenAI, Anthropic, Gemini, …). This
    15	 * backend is NOT vendor inference: it is internal agent-to-agent transport
    16	 * through the Ploinky router (the router's default responder returns an agent
    17	 * capability message, not vendor completion), analogous to the router's own
    18	 * discovery / delegation calls.
    19	 *
    20	 * The assertion's `rch` must bind the EXACT outbound body bytes via
    21	 * `computeRchHttp`, byte-for-byte matching what the router + AgentServer
    22	 * recompute. The Achilles execution path (`createAchillesExecutionHandle` →
    23	 * `callLLMStreaming`) serializes the body internally, so it cannot expose the
    24	 * exact bytes for signing. This backend therefore owns serialization, signing,
    25	 * and the `node:http`/`node:https` POST so the bytes it hashes are the bytes it
    26	 * sends. The response parser accepts both non-streaming OpenAI JSON and SSE,
    27	 * then emits the gateway's normalized-stream contract.
    28	 */

    96	async function resolveVerifiedRouter(ctx, providerRecord) {
    97	    const runtime = await loadVerifiedPloinkyRouterDescriptor({
    98	        env: ctx?.routerDescriptorEnv || process.env,
    99	        loadVerifier: ctx?.loadDescriptorVerifier,
   100	    });
   101	    const configuredOrigin = String(ctx?.env?.PLOINKY_ROUTER_URL || '').trim();
   102	    const providerOrigin = String(providerRecord?.baseUrl || providerRecord?.base_url || '').trim();
   103	    for (const [label, origin] of [
   104	        ['PLOINKY_ROUTER_URL', configuredOrigin],
   105	        ['provider base_url', providerOrigin],
   106	    ]) {
   107	        if (origin && origin.replace(/\/+$/, '') !== runtime.physicalOrigin) {
   108	            throw new Error(`${label} disagrees with the verified Router physical origin`);
   109	        }
   110	    }
   111	    return runtime;
   112	}

   133	/**
   134	 * Build the OpenAI chat-completions request payload from the normalized request
   135	 * and the resolved model. Mirrors the parameter passthrough of the OpenAI
   136	 * backend so a Ploinky agent sees the same request shape any OpenAI-compatible
   137	 * vendor would.
   138	 *
   139	 * @param {object} normalizedReq
   140	 * @param {string} modelId
   141	 * @param {object} settings
   142	 * @returns {object}
   143	 */
   144	function buildPayload(normalizedReq, modelId, settings, { stream = false } = {}) {
   145	    const payload = {
   146	        model: modelId,
   147	        messages: normalizedReq.messages || [],
   148	        stream,
   149	    };
   150	
   151	    if (normalizedReq.max_tokens != null)
   152	        payload.max_tokens = normalizedReq.max_tokens;
   153	    if (normalizedReq.temperature != null)
   154	        payload.temperature = normalizedReq.temperature;
   155	    if (normalizedReq.top_p != null) payload.top_p = normalizedReq.top_p;
   156	    if (normalizedReq.stop != null) payload.stop = normalizedReq.stop;
   157	    if (normalizedReq.tools && normalizedReq.tools.length > 0)
   158	        payload.tools = normalizedReq.tools;
   159	    if (normalizedReq.tool_choice != null)
   160	        payload.tool_choice = normalizedReq.tool_choice;
   161	    if (normalizedReq.response_format != null)
   162	        payload.response_format = normalizedReq.response_format;
   163	    if (settings && settings.extra_body)
   164	        Object.assign(payload, settings.extra_body);
   165	
   166	    return payload;
   167	}

   303	            self: agentId,

   307	        const headers = {
   308	            'Content-Type': 'application/json',
   309	            Authorization: `Bearer ${assertion}`,
   310	            Host: runtime.requestAuthority,
   311	        };

   314	            requestId: ctx.requestId,
   315	            model: modelId,
   316	            subjectId,
   317	        });

   321	            stream,

   326	    classifyError(error, _ctx) {
   327	        const status = getProviderStatus(error);
   328	        const body = error.body || {};
   329	        const errorType = getProviderErrorType(error);
   330	
   331	        if (status === HTTP_STATUS.UNAUTHORIZED) {
   332	            return new ProviderAuthError(
   333	                PROVIDER_LABEL,
   334	                'Agent assertion rejected by router'
   335	            );
   336	        }
   337	        if (status === HTTP_STATUS.FORBIDDEN) {
   338	            return new ProviderAuthError(PROVIDER_LABEL, 'Access denied');
   339	        }
   340	        if (status === HTTP_STATUS.NOT_FOUND) {
   341	            const model =
   342	                body.error?.param === 'model'
   343	                    ? body.error?.message
   344	                    : 'unknown';
   345	            return new ProviderModelNotFoundError(PROVIDER_LABEL, model);
   346	        }
   347	        if (status === HTTP_STATUS.TOO_MANY_REQUESTS) {
   348	            if (
   349	                errorType === 'insufficient_quota' ||
   350	                errorType === 'billing_hard_limit_reached'
   351	            ) {
   352	                return new ProviderQuotaError(PROVIDER_LABEL);
   353	            }
   354	            return new ProviderRateLimitError(PROVIDER_LABEL);
   355	        }
   356	        if (status === HTTP_STATUS.BAD_REQUEST) {
   357	            if (errorType === 'content_policy_violation') {
   358	                return new ProviderContentPolicyError(PROVIDER_LABEL);
   359	            }
   360	        }
   361	        if (status >= HTTP_STATUS.INTERNAL_SERVER_ERROR && status < 600) {
   362	            return classifyTransportOrServerError(PROVIDER_LABEL, error, status);
   363	        }
   364	
   365	        return classifyTransportOrServerError(PROVIDER_LABEL, error);
   366	    },

   461	/**
   462	 * An agent can only report a failure in-band once HTTP 200 has been sent:
   463	 * AgentServer flushes the SSE headers before a command handler starts, and a
   464	 * handler that exits cleanly may still print an error body. Such a payload has
   465	 * an `error` member and no `choices`. It must surface as a thrown, classifiable
   466	 * error rather than as a completion without content.
   467	 *
   468	 * @param {*} parsed
   469	 * @returns {boolean}
   470	 */
   471	function isInBandError(parsed) {
   472	    if (!parsed || typeof parsed !== 'object' || Array.isArray(parsed)) return false;
   473	    if (parsed.error === undefined || parsed.error === null || parsed.error === false) return false;
   474	    return !Array.isArray(parsed.choices) || parsed.choices.length === 0;
   475	}
   476	
   477	/**
   478	 * Build the error thrown for an in-band agent failure. The shape matches the
   479	 * HTTP error branch (`status` plus the parsed `body`), so `classifyError` maps
   480	 * it the same way. A numeric `error.status` or `error.code` in the HTTP error
   481	 * range selects the status; anything else is a bad-gateway failure.
   482	 *
   483	 * @param {object} parsed
   484	 * @returns {Error}
   485	 */
   486	function inBandAgentError(parsed) {
   487	    const detail = parsed.error;
   488	    const info = detail && typeof detail === 'object' ? detail : { message: String(detail) };
   489	    const declared = [info.status, info.code, parsed.status].find(
   490	        (value) => Number.isInteger(value) && value >= 400 && value <= 599
   491	    );
   492	    const message = typeof info.message === 'string' && info.message ? info.message : 'unspecified agent error';
   493	    const err = new Error(`Ploinky agent OpenAI in-band error: ${message}`);
   494	    err.status = declared || HTTP_STATUS.BAD_GATEWAY;
   495	    err.body = { ...parsed, error: info };
   496	    return err;
   497	}

   499	/**
   500	 * Error for an SSE response that ended without `[DONE]` and without any
   501	 * `finish_reason`: the agent stopped mid-answer, so the text received so far
   502	 * is not a completed response.
   503	 *
   504	 * @returns {Error}
   505	 */
   506	function truncatedStreamError() {
   507	    const err = new Error('Ploinky agent OpenAI stream ended before completion');
   508	    err.status = HTTP_STATUS.BAD_GATEWAY;
   509	    err.body = {
   510	        error: {
   511	            type: 'stream_truncated',
   512	            message: 'The agent stream ended without [DONE] or a finish_reason',
   513	        },
   514	    };
   515	    return err;
   516	}

   518	async function* makeOpenAiCompletionStream(url, headers, payload, signal, meta) {
   519	    const response = await doRequest(url, 'POST', headers, payload, signal);
   520	
   521	    if (response.statusCode >= 400) {
   522	        const body = await collectBody(response);
   523	        let parsed = {};
   524	        try {
   525	            parsed = JSON.parse(body);
   526	        } catch {
   527	            /* not JSON */
   528	        }
   529	        const err = new Error(
   530	            `Ploinky agent OpenAI error: ${response.statusCode}`
   531	        );
   532	        err.status = response.statusCode;
   533	        err.body = parsed;
   534	        throw err;
   535	    }
   536	
   537	    const contentType = String(response.headers?.['content-type'] || '');
   538	    if (!contentType.includes('text/event-stream')) {
   539	        const body = await collectBody(response);
   540	        let parsed;
   541	        try {
   542	            parsed = JSON.parse(body || '{}');
   543	        } catch (err) {
   544	            const parseError = new Error(
   545	                `Ploinky agent OpenAI returned invalid JSON: ${err.message}`
   546	            );
   547	            parseError.status = HTTP_STATUS.BAD_GATEWAY;
   548	            parseError.body = {};
   549	            throw parseError;
   550	        }
   551	        if (isInBandError(parsed)) throw inBandAgentError(parsed);
   552	        yield* completionJsonToEvents(parsed, meta);
   553	        return;
   554	    }
   555	
   556	    const state = {};
   557	
   558	    for await (const event of parseSSE(response)) {
   559	        if (event.data === '[DONE]') {
   560	            yield {
   561	                type: 'done',
   562	                data: {
   563	                    finish_reason: state.lastFinishReason || 'stop',
   564	                    model: state.model || meta.model || null,
   565	                },
   566	            };
   567	            return;
   568	        }
   569	
   570	        let parsed;
   571	        try {
   572	            parsed = JSON.parse(event.data);
   573	        } catch {
   574	            continue;
   575	        }
   576	
   577	        // Checked before `message_start`, so a failure reported as the first
   578	        // frame never looks like the beginning of an answer.
   579	        if (isInBandError(parsed)) throw inBandAgentError(parsed);
   580	
   581	        if (!state.started) {
   582	            state.started = true;
   583	            state.model = parsed.model || meta.model || null;
   584	            yield {
   585	                type: 'message_start',
   586	                data: {
   587	                    id: parsed.id || meta.requestId || null,
   588	                    model: state.model,
   589	                    role: 'assistant',
   590	                },
   591	            };
   592	        }
   593	
   594	        for (const choice of parsed.choices || []) {
   595	            const delta = choice.delta || {};
   596	
   597	            if (delta.content) {
   598	                yield { type: 'text_delta', data: { text: delta.content } };
   599	            }
   600	
   601	            if (delta.tool_calls) {
   602	                for (const tc of delta.tool_calls) {
   603	                    yield {
   604	                        type: 'tool_call_delta',
   605	                        data: {
   606	                            index: tc.index ?? 0,
   607	                            id: tc.id || undefined,
   608	                            name: tc.function?.name || undefined,
   609	                            arguments: tc.function?.arguments || undefined,
   610	                        },
   611	                    };
   612	                }
   613	            }
   614	
   615	            if (choice.finish_reason) {
   616	                state.lastFinishReason = choice.finish_reason;
   617	            }
   618	        }
   619	
   620	        if (parsed.usage) {
   621	            yield {
   622	                type: 'usage',
   623	                data: {
   624	                    input_tokens: parsed.usage.prompt_tokens || 0,
   625	                    output_tokens: parsed.usage.completion_tokens || 0,
   626	                    total_tokens: parsed.usage.total_tokens || 0,
   627	                },
   628	            };
   629	        }
   630	    }
   631	
   632	    // The response ended without `[DONE]`. A `finish_reason` means the answer
   633	    // itself completed and only the terminator is missing, so it still counts.
   634	    // Without one the agent stopped mid-answer (for example a command handler
   635	    // that died after writing partial output), which must not pass as success.
   636	    if (!state.lastFinishReason) throw truncatedStreamError();
   637	    yield {
   638	        type: 'done',
   639	        data: {
   640	            finish_reason: state.lastFinishReason,
   641	            model: state.model || meta.model || null,
   642	        },
   643	    };
   644	}

   646	function* completionJsonToEvents(parsed, meta) {
   647	    const choice = Array.isArray(parsed?.choices) ? parsed.choices[0] || {} : {};
   648	    const message = choice.message || {};
   649	    const content = typeof message.content === 'string' ? message.content : '';
   650	    const model = parsed?.model || meta.model || null;
   651	
   652	    yield {
   653	        type: 'message_start',
   654	        data: {
   655	            id: parsed?.id || meta.requestId || null,
   656	            model,
   657	            role: message.role || 'assistant',
   658	        },
   659	    };
   660	
   661	    if (content) {
   662	        yield { type: 'text_delta', data: { text: content } };
   663	    }
   664	
   665	    if (Array.isArray(message.tool_calls)) {
   666	        for (let index = 0; index < message.tool_calls.length; index += 1) {
   667	            const toolCall = message.tool_calls[index] || {};
   668	            yield {
   669	                type: 'tool_call_delta',
   670	                data: {
   671	                    index,
   672	                    id: toolCall.id || undefined,
   673	                    name: toolCall.function?.name || undefined,
   674	                    arguments: toolCall.function?.arguments || undefined,
   675	                },
   676	            };
   677	        }
   678	    }
   679	
   680	    if (parsed?.usage) {
   681	        yield {
   682	            type: 'usage',
   683	            data: {
   684	                input_tokens: parsed.usage.prompt_tokens || 0,
   685	                output_tokens: parsed.usage.completion_tokens || 0,
   686	                total_tokens: parsed.usage.total_tokens || 0,
   687	            },
   688	        };
   689	    }
   690	
   691	    yield {
   692	        type: 'done',
   693	        data: {
   694	            finish_reason: choice.finish_reason || 'stop',
   695	            model,
   696	        },
   697	    };
   698	}

   702	function doRequest(url, method, headers, body, signal) {
   703	    return new Promise((resolve, reject) => {
   704	        const isHttps = url.protocol === 'https:';
   705	        const reqFn = isHttps ? httpsRequest : httpRequest;
   706	
   707	        const opts = {
   708	            hostname: url.hostname,
   709	            port: url.port || (isHttps ? 443 : 80),
   710	            path: url.pathname + url.search,
   711	            method,
   712	            headers: { ...headers, 'Content-Length': Buffer.byteLength(body) },
   713	            signal,
   714	        };
   715	
   716	        const req = reqFn(opts, (res) => resolve(res));
   717	        req.on('error', reject);
   718	        req.write(body);
   719	        req.end();
   720	    });
   721	}
   722	
   723	async function requestJson(url, method, headers, body, signal) {
   724	    const payload = body == null ? null : Buffer.from(body);
   725	    const response = await doRequest(url, method, headers, payload || Buffer.alloc(0), signal);
   726	    const text = await collectBody(response);
   727	    if (response.statusCode >= 400) {
   728	        const err = new Error(`Ploinky agent models error: ${response.statusCode}`);
   729	        err.status = response.statusCode;
   730	        try {
   731	            err.body = JSON.parse(text);
   732	        } catch {
   733	            err.body = {};
   734	        }
   735	        throw err;
   736	    }
   737	    try {
   738	        return JSON.parse(text || '{}');
   739	    } catch (err) {
   740	        throw new Error(`Ploinky agent models returned invalid JSON: ${err.message}`);
   741	    }
   742	}
   743	
   744	function collectBody(res) {
   745	    return new Promise((resolve, reject) => {
   746	        const chunks = [];
   747	        res.on('data', (c) => chunks.push(c));
   748	        res.on('end', () => resolve(Buffer.concat(chunks).toString('utf8')));
   749	        res.on('error', reject);
   750	    });
   751	}

   753	// Server-sent event lines end with LF or CRLF, and one space after a field
   754	// name is optional: `data: [DONE]` and `data:[DONE]` are the same field.
   755	function withoutCarriageReturn(line) {
   756	    return line.endsWith('\r') ? line.slice(0, -1) : line;
   757	}
   758	
   759	function sseField(line, name) {
   760	    if (!line.startsWith(`${name}:`)) return null;
   761	    const value = line.slice(name.length + 1);
   762	    return value.startsWith(' ') ? value.slice(1) : value;
   763	}

   765	async function* parseSSE(res) {
   766	    let buffer = '';
   767	    // One streaming decoder for the whole response, so a multi-byte UTF-8
   768	    // character split across two network chunks is not corrupted.
   769	    const decoder = new TextDecoder('utf-8');
   770	
   771	    for await (const chunk of res) {
   772	        buffer += typeof chunk === 'string' ? chunk : decoder.decode(chunk, { stream: true });
   773	
   774	        const lines = buffer.split('\n');
   775	        buffer = lines.pop(); // keep incomplete line
   776	
   777	        let currentEvent = {};
   778	        for (const rawLine of lines) {
   779	            const line = withoutCarriageReturn(rawLine);
   780	            const data = sseField(line, 'data');
   781	            if (data !== null) {
   782	                currentEvent.data = data;
   783	                yield currentEvent;
   784	                currentEvent = {};
   785	                continue;
   786	            }
   787	            const event = sseField(line, 'event');
   788	            if (event !== null) {
   789	                currentEvent.event = event;
   790	            } else if (line === '') {
   791	                if (currentEvent.data != null) {
   792	                    yield currentEvent;
   793	                    currentEvent = {};
   794	                }
   795	            }
   796	        }
   797	    }
   798	
   799	    buffer += decoder.decode();
   800	    const data = sseField(withoutCarriageReturn(buffer), 'data');
   801	    if (data !== null) {
   802	        yield { data };
   803	    }
   804	}
